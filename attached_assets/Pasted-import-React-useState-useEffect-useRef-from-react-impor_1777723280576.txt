// Funções auxiliares
function calcularDistancia(lat1, lon1, lat2, lon2) {
  const R = 6371000;
  const φ1 = lat1 * Math.PI / 180;
  const φ2 = lat2 * Math.PI / 180;
  const Δφ = (lat2 - lat1) * Math.PI / 180;
  const Δλ = (lon2 - lon1) * Math.PI / 180;
  const a = Math.sin(Δφ/2) * Math.sin(Δφ/2) +
            Math.cos(φ1) * Math.cos(φ2) *
            Math.sin(Δλ/2) * Math.sin(Δλ/2);
  const c = 2 * Math.atan2(Math.sqrt(a), Math.sqrt(1-a));
  return R * c;
}

function formatarDistancia(metros) {
  if (metros < 1000) return `${Math.round(metros)} m`;
  return `${(metros / 1000).toFixed(1)} km`;
}

function FitBounds({ routeCoords }) {
  const map = useMap();
  useEffect(() => {
    if (routeCoords && routeCoords.length > 0) {
      const bounds = L.latLngBounds(routeCoords);
      map.fitBounds(bounds, { padding: [50, 50] });
    }
  }, [routeCoords, map]);
  return null;
}

export default function App() {
  const [currentPosition, setCurrentPosition] = useState(null);
  const [selectedZone, setSelectedZone] = useState(null);
  const [route, setRoute] = useState([]);
  const [distance, setDistance] = useState(null);
  const [duration, setDuration] = useState(null);
  const [isNavigating, setIsNavigating] = useState(false);
  const [loadingRoute, setLoadingRoute] = useState(false);
  const [searchTerm, setSearchTerm] = useState("");
  const [showList, setShowList] = useState(false);
  const mapRef = useRef();

  // Buscar localização
  useEffect(() => {
    navigator.geolocation.getCurrentPosition(
      (pos) => {
        setCurrentPosition({
          lat: pos.coords.latitude,
          lng: pos.coords.longitude,
        });
      },
      (err) => console.error("Erro GPS:", err),
      { enableHighAccuracy: true }
    );
  }, []);

  // Buscar rota real (usando OSRM)
  async function buscarRota(start, end) {
    setLoadingRoute(true);
    try {
      const url = `https://router.project-osrm.org/route/v1/drive/${start.lng},${start.lat};${end.lon},${end.lat}?overview=full&geometries=geojson&steps=true`;
      const response = await fetch(url);
      const data = await response.json();
      
      if (data.routes && data.routes.length > 0) {
        const coordinates = data.routes[0].geometry.coordinates.map(coord => ({
          lat: coord[1],
          lng: coord[0],
        }));
        setRoute(coordinates);
        setDistance(Math.round(data.routes[0].distance));
        setDuration(Math.round(data.routes[0].duration / 60));
        setIsNavigating(true);
      } else {
        alert("Não foi possível calcular rota");
      }
    } catch (error) {
      console.error("Erro:", error);
      alert("Erro ao calcular rota. Verifique internet.");
    } finally {
      setLoadingRoute(false);
    }
  }

  function iniciarNavegacao(zona) {
    setSelectedZone(zona);
    setShowList(false);
    if (currentPosition) {
      buscarRota(currentPosition, {
        lat: zona.lat,
        lng: zona.lon,
      });
    } else {
      alert("Aguardando localização...");
    }
  }

  function pararNavegacao() {
    setIsNavigating(false);
    setRoute([]);
    setSelectedZone(null);
    setDistance(null);
    setDuration(null);
  }

  // Filtrar zonas por busca
  const zonasFiltradas = waypoints.filter(zona =>
    zona.nome.toLowerCase().includes(searchTerm.toLowerCase())
  );

  // Zonas próximas (até 2km)
  const zonasProximas = currentPosition ? waypoints
    .map(zona => ({
      ...zona,
      distancia: calcularDistancia(currentPosition.lat, currentPosition.lng, zona.lat, zona.lon)
    }))
    .filter(zona => zona.distancia < 2000)
    .sort((a, b) => a.distancia - b.distancia)
    .slice(0, 10) : [];

  return (
    <div style={{ height: "100vh", display: "flex", flexDirection: "column" }}>
      {/* Barra de controle */}
      <div style={{ padding: "10px", background: "#1a472a", color: "white", zIndex: 1000 }}>
        <div style={{ display: "flex", gap: "10px", marginBottom: "10px", flexWrap: "wrap" }}>
          <input
            type="text"
            placeholder="Buscar zona (ex: 7922)..."
            value={searchTerm}
            onChange={(e) => setSearchTerm(e.target.value)}
            onFocus={() => setShowList(true)}
            style={{ flex: 2, padding: "10px", borderRadius: "5px", border: "none", fontSize: "16px" }}
          />
          <button
            onClick={() => setShowList(!showList)}
            style={{ padding: "10px 15px", background: "#3498db", color: "white", border: "none", borderRadius: "5px", cursor: "pointer" }}
          >
            📋 Lista
          </button>
          {isNavigating && (
            <button
              onClick={pararNavegacao}
              style={{ padding: "10px 15px", background: "#e74c3c", color: "white", border: "none", borderRadius: "5px", cursor: "pointer" }}
            >
              ⏹️ Parar
            </button>
          )}
        </div>

        {/* Painel de navegação ativa */}
        {isNavigating && selectedZone && (
          <div style={{ background: "#2c5f2d", padding: "10px", borderRadius: "5px" }}>
            <div style={{ display: "flex", justifyContent: "space-between", flexWrap: "wrap", gap: "10px" }}>
              <strong>🎯 Zona {selectedZone.nome}</strong>
              <span>📏 {distance ? formatarDistancia(distance) : "---"}</span>
              <span>⏱️ {duration ? `${duration} min` : "---"}</span>
              {loadingRoute && <span>⏳ Calculando rota...</span>}
            </div>
          </div>
        )}
      </div>

      {/* Mapa */}
      <div style={{ flex: 1, position: "relative" }}>
        <MapContainer
          center={[-22.45, -50.38]}
          zoom={13}
          style={{ height: "100%", width: "100%" }}
          whenReady={(map) => (mapRef.current = map.target)}
        >
          <TileLayer
            url="https://{s}.tile.openstreetmap.org/{z}/{x}/{y}.png"
            attribution='&copy; OpenStreetMap'
          />
          
          {/* Posição atual */}
          {currentPosition && (
            <Marker position={[currentPosition.lat, currentPosition.lng]}>
              <Popup>📍 Você está aqui</Popup>
            </Marker>
          )}
          
          {/* Todas as zonas */}
          {zonasFiltradas.map((zona) => (
            <Marker
              key={zona.nome}
              position={[zona.lat, zona.lon]}
              eventHandlers={{ click: () => iniciarNavegacao(zona) }}
            >
              <Popup>
                <div style={{ textAlign: "center" }}>
                  <strong>🌾 Zona {zona.nome}</strong>
                  <button 
                    onClick={() => iniciarNavegacao(zona)} 
                    style={{ marginTop: "8px", background: "#27ae60", color: "white", border: "none", padding: "5px 10px", borderRadius: "3px", cursor: "pointer", width: "100%" }}
                  >
                    🚜 Navegar
                  </button>
                </div>
              </Popup>
            </Marker>
          ))}
          
          {/* Rota desenhada no mapa */}
          {route.length > 0 && (
            <Polyline
              positions={route.map(p => [p.lat, p.lng])}
              color="#3498db"
              weight={5}
              opacity={0.9}
            />
          )}
          
          <FitBounds routeCoords={route.map(p => [p.lat, p.lng])} />
        </MapContainer>

        {/* LISTA LATERAL DE ZONAS */}
        {showList && (
          <div style={{
            position: "absolute",
            top: 10,
            right: 10,
            bottom: 10,
            width: "300px",
            background: "rgba(0,0,0,0.95)",
            borderRadius: "10px",
            overflow: "hidden",
            display: "flex",
            flexDirection: "column",
            zIndex: 10000,
            boxShadow: "0 4px 20px rgba(0,0,0,0.5)"
          }}>
            <div style={{ padding: "12px", background: "#1a472a", display: "flex", justifyContent: "space-between", alignItems: "center" }}>
              <strong style={{ color: "white" }}>📋 Todas as Zonas ({zonasFiltradas.length})</strong>
              <button onClick={() => setShowList(false)} style={{ background: "#e74c3c", color: "white", border: "none", borderRadius: "5px", padding: "5px 10px", cursor: "pointer" }}>✕ Fechar</button>
            </div>
            
            <div style={{ padding: "10px" }}>
              <input
                type="text"
                placeholder="Filtrar zona..."
                value={searchTerm}
                onChange={(e) => setSearchTerm(e.target.value)}
                style={{ width: "100%", padding: "8px", borderRadius: "5px", border: "1px solid #555", marginBottom: "10px", fontSize: "14px" }}
              />
            </div>
            
            <ScrollView style={{ flex: 1, overflowY: "auto", padding: "10px" }}>
              {zonasFiltradas.map((zona) => {
                const distancia = currentPosition ? calcularDistancia(currentPosition.lat, currentPosition.lng, zona.lat, zona.lon) : null;
                return (
                  <div
                    key={zona.nome}
                    onClick={() => iniciarNavegacao(zona)}
                    style={{
                      padding: "10px",
                      marginBottom: "8px",
                      background: selectedZone?.nome === zona.nome ? "#27ae60" : "#2c3e50",
                      borderRadius: "8px",
                      cursor: "pointer",
                      display: "flex",
                      justifyContent: "space-between",
                      alignItems: "center",
                      transition: "all 0.2s"
                    }}
                  >
                    <div>
                      <strong style={{ color: "white", fontSize: "16px" }}>Zona {zona.nome}</strong>
                      {distancia && distancia < 2000 && (
                        <div style={{ color: "#2ecc71", fontSize: "11px" }}>📍 {formatarDistancia(distancia)}</div>
                      )}
                    </div>
                    <div style={{ color: "#3498db", fontSize: "18px" }}>🚜</div>
                  </div>
                );
              })}
              {zonasFiltradas.length === 0 && (
                <div style={{ textAlign: "center", color: "#aaa", padding: "20px" }}>
                  Nenhuma zona encontrada
                </div>
              )}
            </ScrollView>
            
            {/* Zonas próximas em destaque */}
            {zonasProximas.length > 0 && !searchTerm && (
              <div style={{ padding: "10px", borderTop: "1px solid #444", background: "#1a1a2e" }}>
                <div style={{ color: "#f39c12", fontSize: "12px", marginBottom: "8px" }}>🌟 ZONAS PRÓXIMAS (até 2km)</div>
                {zonasProximas.map((zona) => (
                  <div
                    key={`prox-${zona.nome}`}
                    onClick={() => iniciarNavegacao(zona)}
                    style={{
                      padding: "8px",
                      marginBottom: "5px",
                      background: "#e67e22",
                      borderRadius: "5px",
                      cursor: "pointer",
                      display: "flex",
                      justifyContent: "space-between"
                    }}
                  >
                    <span style={{ fontWeight: "bold" }}>Zona {zona.nome}</span>
                    <span>{formatarDistancia(zona.distancia)}</span>
                  </div>
                ))}
              </div>
            )}
          </div>
        )}
      </div>

      {/* Botão flutuante para abrir lista */}
      {!showList && (
        <button
          onClick={() => setShowList(true)}
          style={{
            position: "absolute",
            bottom: 20,
            right: 20,
            background: "#1a472a",
            color: "white",
            border: "none",
            borderRadius: "50px",
            padding: "12px 20px",
            fontSize: "16px",
            fontWeight: "bold",
            cursor: "pointer",
            zIndex: 1000,
            boxShadow: "0 2px 10px rgba(0,0,0,0.3)"
          }}
        >
          📋 Lista de Zonas
        </button>
      )}
    </div>
  );
}

// Componente ScrollView simples
function ScrollView({ children, style }) {
  return <div style={{ ...style, overflowY: "auto" }}>{children}</div>;
}
